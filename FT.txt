<script>
// 1. FIREBASE CONFIGURATION (Replace this with YOUR keys)
const firebaseConfig = {
    apiKey: "YOUR_API_KEY",
    authDomain: "YOUR_PROJECT.firebaseapp.com",
    projectId: "YOUR_PROJECT_ID",
    storageBucket: "YOUR_PROJECT.appspot.com",
    messagingSenderId: "YOUR_ID",
    appId: "YOUR_APP_ID"
};

// Initialize Firebase
firebase.initializeApp(firebaseConfig);
const db = firebase.firestore();

// State
let accounts = [];
let transactions = [];
const currency = "$";

// Real-time Sync
function initSync() {
    // Sync Accounts
    db.collection("accounts").onSnapshot(snap => {
        accounts = snap.docs.map(doc => ({id: doc.id, ...doc.data()}));
        render();
    });
    // Sync Transactions
    db.collection("transactions").orderBy("date", "desc").onSnapshot(snap => {
        transactions = snap.docs.map(doc => ({id: doc.id, ...doc.data()}));
        render();
    });
}

// Actions
async function addAccount() {
    const name = document.getElementById('accName').value;
    const balance = parseFloat(document.getElementById('accBal').value);
    await db.collection("accounts").add({ name, balance });
    closeModal('accModal');
}

async function addTransaction() {
    const type = document.getElementById('txType').value;
    const accId = document.getElementById('txAcc').value;
    const amount = parseFloat(document.getElementById('txAmt').value);
    const note = document.getElementById('txNote').value;
    const date = document.getElementById('txDate').value || new Date().toISOString().split('T')[0];

    const acc = accounts.find(a => a.id === accId);
    const newBal = type === 'income' ? acc.balance + amount : acc.balance - amount;

    await db.collection("transactions").add({ type, accId, accName: acc.name, amount, note, date });
    await db.collection("accounts").doc(accId).update({ balance: newBal });
    
    closeModal('txModal');
}

// Analysis Logic
function runFinalise() {
    const now = new Date();
    const thirtyDaysAgo = new Date(now.setDate(now.getDate() - 30)).toISOString().split('T')[0];
    
    const recent = transactions.filter(t => t.date >= thirtyDaysAgo && t.type === 'expense');
    
    // 1. Group by Note to find "Maximum Number of Spends"
    const frequency = {};
    recent.forEach(t => frequency[t.note] = (frequency[t.note] || 0) + 1);
    
    const sortedFreq = Object.entries(frequency).sort((a,b) => b[1] - a[1]);
    const topSpend = sortedFreq[0] || ["None", 0];

    // 2. Simple Global-Logic Analysis
    let totalSpent = recent.reduce((s, t) => s + t.amount, 0);
    let advice = "";
    
    if (topSpend[1] > 5) {
        advice = `⚠️ You spent on **${topSpend[0]}** ${topSpend[1]} times this month. High frequency suggests impulsive spending. Try the "24-hour rule" before buying this.`;
    } else {
        advice = `✅ Your spending frequency is well-distributed.`;
    }

    const wastage = recent.filter(t => ["eatout", "subscription", "luxury", "waste"].some(kw => t.note.toLowerCase().includes(kw)));
    const wastageTotal = wastage.reduce((s, t) => s + t.amount, 0);

    if (wastageTotal > (totalSpent * 0.2)) {
        advice += `<br><br>💡 **Growth Plan:** You lost ${currency}${wastageTotal} on non-essentials. If you divert this to an Index Fund (8% avg), in 10 years this month's wastage alone becomes **${currency}${(wastageTotal * 2.15).toFixed(0)}**.`;
    }

    document.getElementById('analysisText').innerHTML = `
        <b>Report (Last 30 Days)</b><br>
        Most Frequent: ${topSpend[0]} (${topSpend[1]} times)<br><br>
        ${advice}
    `;

    showModal('reportModal');
    renderChart(sortedFreq.slice(0, 5));
}

function renderChart(data) {
    const ctx = document.getElementById('chartSpends').getContext('2d');
    if (window.myChart) window.myChart.destroy();
    window.myChart = new Chart(ctx, {
        type: 'bar',
        data: {
            labels: data.map(d => d[0]),
            datasets: [{ label: 'Times Spent', data: data.map(d => d[1]), backgroundColor: '#4F46E5' }]
        }
    });
}

// UI Helpers
function switchTab(tab) {
    document.getElementById('tab-dash').classList.toggle('hidden', tab !== 'dash');
    document.getElementById('tab-ledger').classList.toggle('hidden', tab !== 'ledger');
    document.querySelectorAll('.nb').forEach(b => b.classList.remove('active'));
    event.target.classList.add('active');
}

function showModal(id) { 
    document.getElementById(id).style.display = 'flex';
    if(id === 'txModal') {
        document.getElementById('txAcc').innerHTML = accounts.map(a => `<option value="${a.id}">${a.name}</option>`).join('');
    }
}
function closeModal(id) { document.getElementById(id).style.display = 'none'; }

function render() {
    const net = accounts.reduce((s, a) => s + a.balance, 0);
    document.getElementById('mainBal').innerText = currency + net.toLocaleString();
    
    document.getElementById('accList').innerHTML = accounts.map(a => `
        <div class="tx-item"><span>${a.name}</span><b>${currency}${a.balance.toFixed(2)}</b></div>
    `).join('');

    document.getElementById('txList').innerHTML = transactions.map(t => `
        <div class="tx-item">
            <div>
                <div style="font-weight:600">${t.note} <span class="tag">${t.date}</span></div>
                <small style="color:var(--t2)">${t.accName}</small>
            </div>
            <b style="color:${t.type === 'income' ? 'var(--g)' : 'var(--r)'}">
                ${t.type === 'income' ? '+' : '-'}${currency}${t.amount}
            </b>
        </div>
    `).join('');
}

window.onload = () => {
    initSync();
    document.getElementById('txDate').value = new Date().toISOString().split('T')[0];
};
